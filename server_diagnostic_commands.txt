افتح terminal على السيرفر وشغل الأوامر دي بالترتيب وابعتلي الناتج:

# 1. شوف آخر أخطاء الباك إند
pm2 logs tour-backend --lines 50 --nostream

# 2. تأكد إن الميجريشن اتشغلت فعلاً على قاعدة بيانات السيرفر
cd /var/www/travluyo/backend
npm run migrate

# 3. تأكد من شكل الجدول الفعلي على السيرفر
sudo -u postgres psql -d tour -c "\d blog_posts"
sudo -u postgres psql -d tour -c "\d blog_categories"
