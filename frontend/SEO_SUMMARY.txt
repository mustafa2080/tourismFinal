STEP 3: Integrate SEO Services
───────────────────────────────
في App.jsx، استخدم SEO Service:

import { seoService } from './services/seoService';

// عند تحميل الصفحة:
seoService.updatePageMeta({
  title: 'Page Title',
  description: 'Page Description',
  url: window.location.href
});


STEP 4: Use SEO Hook in Pages
──────────────────────────────
import { useSEO } from './hooks/useSEO';

export default function HomePage() {
  useSEO({
    title: 'Amazing Tours & Travel Packages',
    description: 'Book your dream vacation now',
    keywords: 'tours, travel, packages',
    breadcrumbs: [{ name: 'Home', url: '/' }]
  });

  return <div>{/* content */}</div>;
}


STEP 5: Add Structured Data
────────────────────────────
لكل صفحة، أضف structured data المناسب:

// للصفحة الرئيسية
seoService.updatePageMeta({...});

  price: 1500,
  currency: 'USD',
  rating: 4.5
});

// لصفحة الـ FAQ

  { question: '...', answer: '...' }
]);
